
import sys
import os
import requests
from bs4 import BeautifulSoup
import google.generativeai as genai
import gspread
from datetime import datetime
from PyQt5.QtWidgets import (QApplication, QMainWindow, QWidget, QVBoxLayout,
                             QHBoxLayout, QGroupBox, QLineEdit, QPushButton,
                             QTableWidget, QTextEdit, QTextBrowser, QSplitter,
                             QHeaderView, QTableWidgetItem, QMessageBox)
from PyQt5.QtCore import Qt, QTime

# ==========================================
# 1. API 및 서비스 계정 초기 설정
# ==========================================
GEMINI_API_KEY = "여기에_Gemini_API_키를_입력하세요"
genai.configure(api_key=GEMINI_API_KEY)
model = genai.GenerativeModel('gemini-1.5-flash')

# 현재 폴더 기준 서비스 계정 JSON 경로 지정
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
JSON_PATH = os.path.join(BASE_DIR, "service_account.json")
TARGET_SHEET_NAME = "파이썬용-구글시트01" # 방금 만드신 시트 이름

class SNSAutomationApp(QMainWindow):
    def __init__(self):
        super().__init__()
        self.initUI()
        self.scraped_data = [] # 크롤링한 데이터를 임시 저장할 리스트

    def initUI(self):
        self.setWindowTitle("SNS 소셜 미디어 자동화 대시보드")
        self.setGeometry(100, 100, 1200, 800)

        central_widget = QWidget()
        self.setCentralWidget(central_widget)
        main_layout = QVBoxLayout(central_widget)

        # [상단] RSS 입력 및 제어부
        rss_group = QGroupBox("구글 알리미 RSS 설정 및 제어")
        rss_layout = QHBoxLayout()
        
        self.rss_input = QLineEdit()
        self.rss_input.setPlaceholderText("구글 알리미 RSS 링크를 붙여넣으세요")
        
        self.btn_fetch = QPushButton("1. 뉴스 수집")
        self.btn_ai_summary = QPushButton("2. Gemini 요약")
        self.btn_save = QPushButton("3. 시트 전송")

        rss_layout.addWidget(self.rss_input)
        rss_layout.addWidget(self.btn_fetch)
        rss_layout.addWidget(self.btn_ai_summary)
        rss_layout.addWidget(self.btn_save)
        rss_group.setLayout(rss_layout)
        main_layout.addWidget(rss_group)

        # [중단] 데이터 리스트 & AI 결과창 (좌우 분할)
        splitter = QSplitter(Qt.Horizontal)

        # 좌측: 기사 목록 표
        list_group = QGroupBox("크롤링된 기사 목록")
        list_layout = QVBoxLayout()
        self.table = QTableWidget(0, 3)
        self.table.setHorizontalHeaderLabels(["작성일", "기사 제목", "원문 URL"])
        self.table.horizontalHeader().setSectionResizeMode(1, QHeaderView.Stretch)
        list_layout.addWidget(self.table)
        list_group.setLayout(list_layout)
        splitter.addWidget(list_group)

        # 우측: AI 결과창
        ai_group = QGroupBox("Gemini AI 요약 및 숏폼 스크립트")
        ai_layout = QVBoxLayout()
        self.text_edit = QTextEdit()
        ai_layout.addWidget(self.text_edit)
        ai_group.setLayout(ai_layout)
        splitter.addWidget(ai_group)

        splitter.setSizes([500, 700])
        main_layout.addWidget(splitter, stretch=1)

        # [하단] 로그 창
        log_group = QGroupBox("시스템 로그")
        log_layout = QVBoxLayout()
        self.log_browser = QTextBrowser()
        log_layout.addWidget(self.log_browser)
        log_group.setLayout(log_layout)
        log_group.setMaximumHeight(200)
        main_layout.addWidget(log_group)

        # 버튼 이벤트 연결
        self.btn_fetch.clicked.connect(self.fetch_rss_data)
        self.btn_ai_summary.clicked.connect(self.generate_ai_summary)
        self.btn_save.clicked.connect(self.save_to_google_sheet)

        self.log_message("시스템 초기화 완료. 준비되었습니다.")

    def log_message(self, msg):
        """시간과 함께 로그를 출력합니다."""
        current_time = QTime.currentTime().toString("hh:mm:ss")
        self.log_browser.append(f"[{current_time}] {msg}")
        QApplication.processEvents() # UI 멈춤 방지

    def fetch_rss_data(self):
        rss_url = self.rss_input.text().strip()
        if not rss_url:
            self.log_message("오류: RSS 링크를 입력해주세요.")
            return

        self.log_message("구글 알리미 RSS 데이터 수집 시작...")
        try:
            response = requests.get(rss_url)
            soup = BeautifulSoup(response.content, 'xml')
            entries = soup.find_all('entry')

            self.table.setRowCount(0)
            self.scraped_data = []

            for entry in entries[:5]: # 최신 기사 5개만 가져오기
                title = BeautifulSoup(entry.title.text, "html.parser").text # 구글 알리미 태그 제거
                link = entry.link['href']
                published = datetime.now().strftime("%Y-%m-%d %H:%M") # 현재 수집 시간 기준

                self.scraped_data.append({"title": title, "link": link, "date": published})
                
                # 테이블에 추가
                row = self.table.rowCount()
                self.table.insertRow(row)
                self.table.setItem(row, 0, QTableWidgetItem(published))
                self.table.setItem(row, 1, QTableWidgetItem(title))
                self.table.setItem(row, 2, QTableWidgetItem(link))

            self.log_message(f"총 {len(self.scraped_data)}개의 최신 기사를 가져왔습니다.")
        except Exception as e:
            self.log_message(f"수집 실패: {e}")

    def generate_ai_summary(self):
        if not self.scraped_data:
            self.log_message("오류: 먼저 뉴스를 수집해주세요.")
            return

        self.log_message("Gemini AI 요약 생성 중... (잠시만 기다려주세요)")
        
        # 첫 번째 기사를 요약 샘플로 사용
        target_article = self.scraped_data[0]
        prompt = f"""
        다음 기사 제목과 내용을 바탕으로 SNS 숏폼 대본과 해시태그를 작성해 줘.
        제목: {target_article['title']}
        링크: {target_article['link']}
        """

        try:
            response = model.generate_content(prompt)
            self.text_edit.setText(response.text)
            self.log_message("AI 스크립트 생성이 완료되었습니다.")
        except Exception as e:
            self.log_message(f"Gemini API 오류: {e}")

    def save_to_google_sheet(self):
        ai_result = self.text_edit.toPlainText().strip()
        if not self.scraped_data or not ai_result:
            self.log_message("오류: 저장할 데이터나 AI 요약 결과가 없습니다.")
            return

        self.log_message("구글 시트에 데이터 전송 중...")
        try:
            gc = gspread.service_account(filename=JSON_PATH)
            sh = gc.open(TARGET_SHEET_NAME).sheet1
            
            target_article = self.scraped_data[0]
            
            # 구글 시트에 행 추가 (A열: 제목, B열: AI요약, C열: 링크, D열: 날짜)
            sh.append_row([
                target_article['title'], 
                ai_result, 
                target_article['link'], 
                target_article['date']
            ])
            self.log_message("✅ 구글 시트에 성공적으로 저장되었습니다!")
            QMessageBox.information(self, "성공", "구글 시트 저장이 완료되었습니다.")
        except Exception as e:
            self.log_message(f"구글 시트 저장 실패: {e}")
            self.log_message("※ 주의: service_account.json 파일 위치와 시트 공유 권한(편집자)을 다시 확인해주세요.")

if __name__ == '__main__':
    app = QApplication(sys.argv)
    ex = SNSAutomationApp()
    ex.show()
    sys.exit(app.exec_())
