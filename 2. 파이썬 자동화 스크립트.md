import os
import time
import feedparser
import requests
from bs4 import BeautifulSoup
import google.generativeai as genai
import gspread
from oauth2client.service_account import ServiceAccountCredentials
from datetime import datetime
import pytz
from dotenv import load_dotenv

# 환경변수 로드 (.env 파일)
load_dotenv()

class NewsSocialAutomator:
    def __init__(self, json_key_path, sheet_name, worksheet_name="시트1"):
        """초기 세팅: API 키 및 구글 시트 연동"""
        self.setup_gemini()
        self.sheet = self.setup_google_sheets(json_key_path, sheet_name, worksheet_name)
        
        # 한국 시간대 설정
        self.kst = pytz.timezone('Asia/Seoul')

    def setup_gemini(self):
        """Gemini 1.5 Flash 모델 초기화"""
        api_key = os.getenv("GEMINI_API_KEY")
        if not api_key:
            raise ValueError("GEMINI_API_KEY가 .env 파일에 없습니다.")
        genai.configure(api_key=api_key)
        self.model = genai.GenerativeModel('gemini-1.5-flash')

    def setup_google_sheets(self, json_key_path, sheet_name, worksheet_name):
        """구글 시트 API 인증 및 시트 객체 반환"""
        scope = [
            "https://spreadsheets.google.com/feeds",
            "https://www.googleapis.com/auth/drive"
        ]
        try:
            creds = ServiceAccountCredentials.from_json_keyfile_name(json_key_path, scope)
            client = gspread.authorize(creds)
            return client.open(sheet_name).worksheet(worksheet_name)
        except Exception as e:
            print(f"구글 시트 연동 실패: {e}")
            raise

    def fetch_rss_links(self, rss_url, max_items=3):
        """RSS 피드에서 최신 기사 URL 및 제목 추출"""
        print(f"[{rss_url}] RSS 파싱 중...")
        feed = feedparser.parse(rss_url)
        articles = []
        for entry in feed.entries[:max_items]:
            articles.append({
                'title': entry.title,
                'link': entry.link
            })
        return articles

    def scrape_article_text(self, url):
        """기사 본문 스크래핑 (광고 및 불필요한 태그 제거)"""
        try:
            headers = {'User-Agent': 'Mozilla/5.0'}
            response = requests.get(url, headers=headers, timeout=10)
            response.raise_for_status()
            
            soup = BeautifulSoup(response.text, 'html.parser')
            
            # 본문 추출 (일반적인 p 태그 중심 추출)
            paragraphs = soup.find_all('p')
            content = " ".join([p.get_text(strip=True) for p in paragraphs if p.get_text(strip=True)])
            return content
        except Exception as e:
            print(f"스크래핑 에러 ({url}): {e}")
            return None

    def generate_sns_content(self, text_content):
        """Gemini AI를 활용하여 요약 및 숏폼 대본 생성"""
        prompt = f"""
        아래 기사 내용을 바탕으로 1) 핵심 요약과 2) SNS에 포스팅할 숏폼용 텍스트를 작성해 줘.
        
        [본문 내용]
        {text_content[:5000]} # 토큰 최적화를 위해 길이 제한
        """
        try:
            response = self.model.generate_content(prompt)
            return response.text
        except Exception as e:
            print(f"Gemini API 에러: {e}")
            return "AI 요약 실패"

    def save_to_sheet(self, title, summary, url):
        """구글 시트에 [제목, 요약문, URL, 작성일] 형태로 저장"""
        current_kst_time = datetime.now(self.kst).strftime("%Y-%m-%d %H:%M")
        row_data = [title, summary, url, current_kst_time]
        
        try:
            self.sheet.append_row(row_data)
            print(f"✅ 시트 저장 완료: {title}")
        except Exception as e:
            print(f"시트 저장 에러: {e}")

    def run_pipeline(self, rss_urls):
        """전체 자동화 파이프라인 실행"""
        for rss in rss_urls:
            articles = self.fetch_rss_links(rss)
            
            for article in articles:
                print(f"\n▶ 처리 중: {article['title']}")
                
                # 1. 웹 스크래핑
                content = self.scrape_article_text(article['link'])
                if not content or len(content) < 100:
                    print("  -> 본문이 너무 짧거나 추출에 실패하여 건너뜁니다.")
                    continue
                
                # 2. AI 요약
                print("  -> AI 요약 및 대본 작성 중...")
                sns_script = self.generate_sns_content(content)
                
                # 3. 시트 저장
                self.save_to_sheet(article['title'], sns_script, article['link'])
                
                # 차단 방지를 위한 딜레이
                time.sleep(3)


if __name__ == "__main__":
    # --- 설정 영역 ---
    # 다운받은 구글 클라우드 JSON 키 파일 이름
    JSON_KEY_FILE = "service_account.json" 
    
    # 구글 시트 파일 이름 (공유 설정 필수)
    SHEET_TITLE = "SNS 뉴스스크랩" 
    
    # 크롤링할 대상 RSS 주소 목록 (예: 연합뉴스, 구글 뉴스 등)
    TARGET_RSS_URLS = [
        "https://www.yonhapnewstv.co.kr/category/news/politics/feed/" 
    ]
    
    # --- 실행 영역 ---
    print("🚀 뉴스 요약 SNS 자동화 봇 실행 시작")
    automator = NewsSocialAutomator(
        json_key_path=JSON_KEY_FILE, 
        sheet_name=SHEET_TITLE
    )
    
    automator.run_pipeline(TARGET_RSS_URLS)
    print("\n🎉 모든 파이프라인 처리가 완료되었습니다!")

###💡 코드 활용 포인트

1. PyQt5 통합 용이성: 메인 로직이 NewsSocialAutomator라는 클래스로 묶여 있기 때문에, 
PyQt5 대시보드 파일에서 from main import NewsSocialAutomator로 불러와 버튼 클릭 이벤트(Signals & Slots)나 QThread 백그라운드 작업에 쉽게 연결하실 수 있습니다.

2. 시간 포맷 완벽 적용: pytz 라이브러리를 사용하여 서버가 어디에 있든 정확한 한국 표준시(KST)로 YYYY-MM-DD HH:mm 형태의 작성일이 구글 시트 D열에 찍히도록 구현했습니다.

3. 오류 방어: 기사 본문 구조가 달라서 크롤링에 실패하더라도 프로그램이 뻗지 않도록 try-except로 꼼꼼하게 예외 처리를 해두었습니다.
